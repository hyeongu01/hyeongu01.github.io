## 이진 탐색 트리란?

- 각 노드는 최대 두 개의 자식 노드를 가지게 된다.
- **왼쪽 자식**은 부모보다 **작은 값**, **오른쪽 자식**은 부모보다 **크게** 된다.

---

## Node 및 Tree 구조 구현

### Node 클래스

```swift
class Node {
    let data: Int
    var left: Node?
    var right: Node?

    init(_ data: Int) {
        self.data = data
    }
}
```

### Tree 구조체

```swift
struct Tree {
    var root: Node? = nil

    func isEmpty() -> Bool { root == nil }

    mutating func insert(_ newData: Int) { ... }

    func preOrder(...), inOrder(...), postOrder(...) { ... }
}
```

---

## 삽입 연산 (Insert)

루트 노드가 없으면 새 노드를 루트로 설정하고,
기타 경우에는 트리의 왼쪽 또는 오른쪽으로 적절히 내려가면서 위치를 찾아 삽임.

```swift
mutating func insert(_ newData: Int) {
    let newNode = Node(newData)
    if isEmpty() {
        root = newNode
    } else {
        var current = root!
        while true {
            var next: Node? = nil
            if current.data >= newData {
                next = current.left
            } else {
                next = current.right
            }
            guard let nextNode = next else {
                break
            }
            current = nextNode
        }
        if current.data >= newData {
            current.left = newNode
        } else {
            current.right = newNode
        }
    }
}
```

## 트리 순회 방법 구현

| 순회 방식 | 순서                 |
| --------- | -------------------- |
| 전위 순회 | 루트 → 왼쪽 → 오른쪽 |
| 중위 순회 | 왼쪽 → 루트 → 오른쪽 |
| 후위 순회 | 왼쪽 → 오른쪽 → 루트 |

```swift
func preOrder(_ res: inout [Int], parent: Node?) {
	guard let parentNode = parent else {
		return
	}
	res.append(parentNode.data)
	preOrder(&res, parent: parentNode.left)
	preOrder(&res, parent: parentNode.right)
}

func postOrder(_ res: inout [Int], parent: Node?) {
	guard let parentNode = parent else {
		return
	}
	postOrder(&res, parent: parentNode.left)
	postOrder(&res, parent: parentNode.right)
	res.append(parentNode.data)
}

func inOrder(_ res: inout [Int], parent: Node?) {
	guard let parentNode = parent else {
		return
	}
	inOrder(&res, parent: parentNode.left)
	res.append(parentNode.data)
	inOrder(&res, parent: parentNode.right)
}
```

## 테스트 코드

```swift
var tree = Tree()

let valuesToInsert = [7, 3, 9, 1, 5, 8, 10]
for value in valuesToInsert {
    tree.insert(value)
}

var preorderResult: [Int] = []
var inorderResult: [Int] = []
var postorderResult: [Int] = []

tree.preOrder(&preorderResult, parent: tree.root)
tree.inOrder(&inorderResult, parent: tree.root)
tree.postOrder(&postorderResult, parent: tree.root)

print("전위 순회 결과: \(preorderResult)")   // [7, 3, 1, 5, 9, 8, 10]
print("중위 순회 결과: \(inorderResult)")   // [1, 3, 5, 7, 8, 9, 10]
print("후위 순회 결과: \(postorderResult)") // [1, 5, 3, 8, 10, 9, 7]

```

## 정리 및 배우는 점

- Swift에서 `class`는 **참조 타입** 이기 때문에 트리 구조 구현에 적합함.
- 트리 순회는 **재귀 함수**로 구현하면 구조가 명확하고 직관적이다.
	- 다만, 함수를 재귀적으로 호출하기 때문에 스택 오버플로우를 주의!
	- 명시적 스택을 생성해 순회하는 방식은 스택 오버플로우를 방지할 수 있다.

## 전체 코드

```swift
//
//  main.swift
//  programmers
//
//  Created by 최현우 on 7/1/25.
//

import Foundation

class Node {
    let data: Int
    var left: Node?
    var right: Node?
    
    init(_ data: Int) {
        self.data = data
        self.left = nil
        self.right = nil
    }
}

struct Tree {
    var root: Node? = nil
    
    func isEmpty() -> Bool { root == nil }
    
    mutating func insert(_ newData: Int) {
        let newNode = Node(newData)
        if isEmpty() {
            root = newNode
        } else {
            var current = root!
            while true {
                var next: Node? = nil
                
                if current.data >= newData {
                    next = current.left
                } else {
                    next = current.right
                }
                guard let nextNode = next else {
                    break
                }
                current = nextNode
            }
            if current.data >= newData {
                current.left = newNode
            } else {
                current.right = newNode
            }
        }
    }
    
    func preOrder(_ res: inout [Int], parent: Node?) {
        guard let parentNode = parent else {
            return
        }
        res.append(parentNode.data)
        preOrder(&res, parent: parentNode.left)
        preOrder(&res, parent: parentNode.right)
    }
    
    func postOrder(_ res: inout [Int], parent: Node?) {
        guard let parentNode = parent else {
            return
        }
        postOrder(&res, parent: parentNode.left)
        postOrder(&res, parent: parentNode.right)
        res.append(parentNode.data)
    }
    
    func inOrder(_ res: inout [Int], parent: Node?) {
        guard let parentNode = parent else {
            return
        }
        inOrder(&res, parent: parentNode.left)
        res.append(parentNode.data)
        inOrder(&res, parent: parentNode.right)
    }
    
}

var tree = Tree()

let valuesToInsert = [7, 3, 9, 1, 5, 8, 10]
for value in valuesToInsert {
    tree.insert(value)
}

var preorderResult: [Int] = []
var inorderResult: [Int] = []
var postorderResult: [Int] = []

tree.preOrder(&preorderResult, parent: tree.root)
tree.inOrder(&inorderResult, parent: tree.root)
tree.postOrder(&postorderResult, parent: tree.root)

print("전위 순회 결과: \(preorderResult)")   // [7, 3, 1, 5, 9, 8, 10]
print("중위 순회 결과: \(inorderResult)")   // [1, 3, 5, 7, 8, 9, 10]
print("후위 순회 결과: \(postorderResult)") // [1, 5, 3, 8, 10, 9, 7]

```
