## 개요
- 숫자 맞추기 게임을 한다고 하자. (1 ~ 100 사이의 값)
- 숫자를 제시하여 정답과 비교한 결과를 알려준다.
- 어떻게 숫자를 제시해야 효과적일까?


## 해결책
- 1 부터 100까지 순서대로 물어보는 경우 최대 100번의 질문을 해야한다.
- 거꾸로의 경우도 마찬가지!
- 효율적인 방법이 없을까?

> 후보 숫자 범위의 중간값을 물어보는 것을 반복하여 후보지를 좁히자!


## 이진 탐색 구현
> 우선 타겟이 배열에 없는 경우를 예외처리 하기 위해 에러 타입을 지정해준다.

```swift
// MARK: - 에러 타입
enum SearchError: Error {
    case NumberIsNotFounded
}
```

> 이진 탐색 함수: `binarySearch(in: for: )`

```swift
// MARK: - 이진 탐색 함수 구현
// 정답의 인덱스를 반환
func binarySearch(in array: [Int], for target : Int) throws -> Int {
    var bottom = 0
    var top = array.count - 1
    
    while bottom <= top {
        let mid = bottom + (top - bottom) / 2
        let guess = array[mid]
        
        #if DEBUG
        print("guess: \(guess)")
        #endif
        
        if guess > target {
            top = mid - 1
        } else if guess < target {
            bottom = mid + 1
        } else {
            return mid
        }
    }
    
    throw SearchError.NumberIsNotFounded
}
```


## 단순한 탐색 방식과 비교
> 비교를 위한 단순 탐색 함수: `simpleSearch(in: , for: )`

```swift
// MARK: - 단순한 방식: 처음부터 끝까지 대입
func simpleSearch(in array: [Int], for target: Int) throws -> Int {
    for index in 0..<array.count {
        let guess = array[index]
        
        #if DEBUG
        print("guess: \(guess)")
        #endif
        
        if guess == target {
            return index
        }
    }
    throw SearchError.NumberIsNotFounded
}
```

## 결론
코드 자체는 simpleSearch 함수가 단순했지만, 대입 횟수를 비교하면, 숫자의 범위가 커질수록 binarySearch 의 효율이 빛을 발한다.

## 전체 코드
```swift
import Foundation

// MARK: - SearchError.swift
enum SearchError: Error {
    case NumberIsNotFounded
}

// MARK: - binarySearch.swift
// 정답의 인덱스를 반환
func binarySearch(in array: [Int], for target : Int) throws -> Int {
    var bottom = 0
    var top = array.count - 1
    var guessCount = 0
    
    while bottom <= top {
        let mid = bottom + (top - bottom) / 2
        let guess = array[mid]
        guessCount += 1
        
        if guess > target {
            top = mid - 1
        } else if guess < target {
            bottom = mid + 1
        } else {
            print("binarySearch find \(target) in \(guessCount) guesses")
            return mid
        }
    }
    throw SearchError.NumberIsNotFounded
}

// MARK: - simpleSearch.swift
func simpleSearch(in array: [Int], for target: Int) throws -> Int {
    var guessCount = 0
    
    for index in 0..<array.count {
        let guess = array[index]
        guessCount += 1
        
        if guess == target {
            print("simpleSearch find \(target) in \(guessCount) guesses")
            return index
        }
    }
    throw SearchError.NumberIsNotFounded
}

// MARK: - main.swift
// 샘플 데이터
let sampleArrays: [[Int]] = [
    Array(1...10),
    Array(1...20),
    Array(1...50),
    Array(1...100),
    Array(1...1000)
]

let sampleTargets: [Int] = [7, 16, 37, 88, 991]

// 비교 실행
for i in 0..<sampleArrays.count {
    let array = sampleArrays[i]
    let target = sampleTargets[i]
    
    print("\n=== 샘플 \(i+1): 배열 크기 \(array.count), 찾을 값 \(target) ===")
    _ = try simpleSearch(in: array, for: target)
    _ = try binarySearch(in: array, for: target)
}
```
